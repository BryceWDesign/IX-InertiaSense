cmake_minimum_required(VERSION 3.16)
project(IX_InertiaSense VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ix_inertiasense STATIC src/core.cpp)
target_include_directories(ix_inertiasense PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(MSVC)
 target_compile_options(ix_inertiasense PRIVATE /W4)
else()
 target_compile_options(ix_inertiasense PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(inertiasense src/cli.cpp)
target_link_libraries(inertiasense PRIVATE ix_inertiasense)
include(CTest)
if(BUILD_TESTING)
 add_executable(ix_core_tests tests/test_core.cpp)
 target_link_libraries(ix_core_tests PRIVATE ix_inertiasense)
 add_test(NAME ix_core_tests COMMAND ix_core_tests)
endif()
install(TARGETS ix_inertiasense inertiasense)
install(DIRECTORY include/ DESTINATION include)
